Q1. Implement K-means from scratch

import numpy as np

def kmeans(X, k=3, max_iter=100, tol=1e-4):
    n, d = X.shape
    # Init: pick k random samples as centroids
    idx = np.random.choice(n, k, replace=False)
    C = X[idx].copy()

    for _ in range(max_iter):
        # Assign each x to nearest centroid
        dists = np.linalg.norm(X[:, None] - C[None, :], axis=2)
        labels = dists.argmin(axis=1)
        # Update centroids
        new_C = np.array([X[labels==j].mean(axis=0) for j in range(k)])
        if np.linalg.norm(new_C - C) < tol:
            break
        C = new_C
    return labels, C

# Discussion: init strategy (K-means++), handling empty cluster

Q2. Implement logistic regression with (batch) gradient descent

def sigmoid(z): return 1 / (1 + np.exp(-z))

def train_logreg(X, y, lr=0.01, epochs=100):
    n, d = X.shape
    w = np.zeros(d)
    b = 0
    for _ in range(epochs):
        z = X @ w + b
        p = sigmoid(z)
        grad_w = X.T @ (p - y) / n
        grad_b = (p - y).mean()
        w -= lr * grad_w
        b -= lr * grad_b
    return w, b

# Discussion: bias term, learning rate, regularization L2

Q3. Compute AUC-ROC from scratch

Cơ chế: AUC-ROC = diện tích dưới đường cong ROC (trục X = FPR, trục Y = TPR) khi thử TẤT CẢ ngưỡng phân loại có thể từ cao xuống thấp. Thay vì thử từng ngưỡng rồi vẽ đường cong rồi tích phân (chậm), thuật toán dưới đây khai thác 1 mẹo: sắp xếp sample theo prob GIẢM DẦN rồi ĐI QUA TỪNG SAMPLE 1 LẦN — mỗi khi gặp sample dương (y=1) thì TP tăng 1 (đường cong "đi lên"), gặp sample âm (y=0) thì FP tăng 1 (đường cong "đi ngang"). Diện tích dưới đường cong = cộng dồn diện tích HÌNH THANG (trapezoidal rule) nhỏ ở mỗi bước — với đường cong dạng bậc thang (không phải cong mượt), hình thang ở mỗi bước ngang thực chất suy biến thành hình chữ nhật (chiều cao = TPR hiện tại, chiều rộng = ΔFPR của bước đó).

def auc_roc(y_true, y_prob):
    # Sort by predicted prob desc
    order = np.argsort(-y_prob)
    y_true = y_true[order]

    pos = y_true.sum()
    neg = len(y_true) - pos
    tp = fp = 0
    prev_fpr = 0
    auc = 0
    for yi in y_true:
        if yi == 1: tp += 1
        else: fp += 1
        tpr = tp / pos
        fpr = fp / neg
        # Trapezoidal area
        auc += (fpr - prev_fpr) * tpr
        prev_fpr = fpr
    return auc

# Discussion: interpret AUC 0.5 = random, 1.0 = perfect

Q6. Cross-entropy loss + gradient

def softmax(z):
    z = z - z.max(axis=-1, keepdims=True)  # numeric stability
    e = np.exp(z)
    return e / e.sum(axis=-1, keepdims=True)

def cross_entropy(logits, y_true):
    """y_true one-hot."""
    p = softmax(logits)
    return -np.sum(y_true * np.log(p + 1e-12), axis=-1).mean()

def ce_grad(logits, y_true):
    """dL/dlogits = softmax(logits) - y_true / batch."""
    p = softmax(logits)
    return (p - y_true) / len(y_true)

Q7. Implement dropout

def dropout(x, p, training=True):
    if not training or p == 0:
        return x
    mask = (np.random.rand(*x.shape) > p).astype(float)
    return x * mask / (1 - p)  # inverted scaling

Q8. LayerNorm implementation

class LayerNorm(nn.Module):
    def __init__(self, dim, eps=1e-5):
        super().__init__()
        self.gamma = nn.Parameter(torch.ones(dim))
        self.beta = nn.Parameter(torch.zeros(dim))
        self.eps = eps

    def forward(self, x):
        mean = x.mean(-1, keepdim=True)
        var = x.var(-1, keepdim=True, unbiased=False)
        x_hat = (x - mean) / torch.sqrt(var + self.eps)
        return self.gamma * x_hat + self.beta

class AttentionHead(nn.Module):
    def __init__(self, d_model, d_head):
        super().__init__()
        self.Wq = nn.Linear(d_model, d_head, bias=False)
        self.Wk = nn.Linear(d_model, d_head, bias=False)

Q10. Positional encoding sinusoidal

def positional_encoding(seq_len, d_model):
    pe = np.zeros((seq_len, d_model))
    position = np.arange(seq_len)[:, None]
    div = np.exp(np.arange(0, d_model, 2) * -(np.log(10000) / d_model))
    pe[:, 0::2] = np.sin(position * div)
    pe[:, 1::2] = np.cos(position * div)
    return pe

Q11. Simple LSTM cell

class LSTMCell(nn.Module):
    def __init__(self, d_in, d_h):
        super().__init__()
        self.W_f = nn.Linear(d_in + d_h, d_h)
        self.W_i = nn.Linear(d_in + d_h, d_h)
        self.W_o = nn.Linear(d_in + d_h, d_h)
        self.W_c = nn.Linear(d_in + d_h, d_h)

    def forward(self, x, h, c):
        combined = torch.cat([x, h], dim=-1)
        f = torch.sigmoid(self.W_f(combined))
        i = torch.sigmoid(self.W_i(combined))
        o = torch.sigmoid(self.W_o(combined))
        c_tilde = torch.tanh(self.W_c(combined))
        c_new = f * c + i * c_tilde
        h_new = o * torch.tanh(c_new)
        return h_new, c_new

Q12. Batch normalization

class BatchNorm1d(nn.Module):
    def __init__(self, dim, momentum=0.1, eps=1e-5):
        super().__init__()
        self.gamma = nn.Parameter(torch.ones(dim))
        self.beta = nn.Parameter(torch.zeros(dim))
        self.register_buffer('running_mean', torch.zeros(dim))
        self.register_buffer('running_var', torch.ones(dim))
        self.momentum = momentum
        self.eps = eps

    def forward(self, x):
        if self.training:
            mean = x.mean(0)
            var = x.var(0, unbiased=False)
            self.running_mean = (1-self.momentum)*self.running_mean + self.momentum*mean.detach()
            self.running_var = (1-self.momentum)*self.running_var + self.momentum*var.detach()
        else:
            mean, var = self.running_mean, self.running_var
        x_hat = (x - mean) / torch.sqrt(var + self.eps)
        return self.gamma * x_hat + self.beta

Q13. BPE tokenizer training step

BPE (Byte-Pair Encoding) = cách tách câu thành token không dựa theo từ điển cố định, mà học từ data: bắt đầu từ từng ký tự riêng lẻ, lặp lại việc "gộp cặp ký tự xuất hiện cạnh nhau nhiều nhất thành 1 token mới" hàng nghìn lần. Kết quả: từ phổ biến ("the") thành 1 token, từ hiếm/từ mới ("predictoring") bị tách thành nhiều mảnh con quen thuộc — không bao giờ gặp "unknown token" hoàn toàn.

from collections import Counter

def train_bpe_step(word_freqs):
    """Given word freq dict → 1 merge step."""
    # Count adjacent pair frequencies
    pairs = Counter()
    for word, freq in word_freqs.items():
        symbols = word.split()
        for i in range(len(symbols) - 1):
            pairs[(symbols[i], symbols[i+1])] += freq
    # Merge most frequent
    top = pairs.most_common(1)[0][0]
    new_word_freqs = {}
    for word, freq in word_freqs.items():
        new_word = word.replace(f"{top[0]} {top[1]}", f"{top[0]}{top[1]}")
        new_word_freqs[new_word] = freq
    return new_word_freqs, top

Q14. Top-P (nucleus) sampling

def top_p_sample(logits, p=0.9):
    probs = np.exp(logits) / np.exp(logits).sum()
    order = np.argsort(-probs)
    cumsum = probs[order].cumsum()
    # Find cutoff index where cumsum >= p
    cutoff = np.searchsorted(cumsum, p) + 1
    keep = order[:cutoff]
    # Renormalize
    keep_probs = probs[keep] / probs[keep].sum()
    return np.random.choice(keep, p=keep_probs)

Q15. Cosine similarity for embeddings

def cosine_similarity(a, b):
    return np.dot(a, b) / (np.linalg.norm(a) * np.linalg.norm(b) + 1e-9)

def semantic_search(query_emb, corpus_embs, top_k=5):
    sims = np.array([cosine_similarity(query_emb, e) for e in corpus_embs])
    top_idx = np.argsort(-sims)[:top_k]
    return top_idx, sims[top_idx]

Q16. Simple LoRA adapter

LoRA (Low-Rank Adaptation): đóng băng (freeze) toàn bộ weight gốc base_layer, chỉ train thêm 2 ma trận nhỏ B (d_in × rank) và A (rank × d_out) với rank rất nhỏ (8, 16...) so với d_in/d_out (hàng nghìn). Output = output gốc + (x qua đường tắt B→A) × hệ số scale. Vì rank nhỏ, số tham số cần train giảm hàng trăm-nghìn lần so với fine-tune toàn bộ — đây là lý do file LoRA chỉ ~50MB thay vì vài GB (liên hệ Multi-LoRA serving ở L7.2 mục 3).

class LoRA(nn.Module):
    def __init__(self, base_layer, rank=8, alpha=16):
        super().__init__()
        self.base = base_layer  # frozen
        for p in self.base.parameters(): p.requires_grad = False
        d_in = base_layer.in_features
        d_out = base_layer.out_features
        self.A = nn.Parameter(torch.zeros(rank, d_out))
        self.B = nn.Parameter(torch.randn(d_in, rank) * 0.01)
        self.scale = alpha / rank

    def forward(self, x):
        return self.base(x) + (x @ self.B @ self.A) * self.scale

Q17. KV cache implementation

KV cache: khi LLM sinh token kế tiếp, nó cần Key/Value của TẤT CẢ token trước đó để tính attention — nếu tính lại từ đầu mỗi lần sinh 1 token mới thì cực tốn (O(n²) cho cả câu). KV cache lưu lại K, V đã tính của các token cũ, mỗi bước chỉ tính thêm K/V của token mới rồi nối (concat) vào cache — đổi bộ nhớ GPU lấy tốc độ. Đây chính là thứ PagedAttention (L7.2 mục 3) tối ưu cách cấp phát.

Q18. Prompt injection sanitizer

import re

INJECTION_PATTERNS = [
    r"ignore\s+(previous|above|prior)\s+instructions?",
    r"disregard\s+(prompts?|system\s+prompt)",
    r"you\s+are\s+now",
    r"new\s+instructions?:",
]

def sanitize(text: str) -> str:
    for pattern in INJECTION_PATTERNS:
        text = re.sub(pattern, "[FILTERED]", text, flags=re.IGNORECASE)
    return text

# Additional: wrap in role tags
def safe_wrap(text: str) -> str:
    cleaned = sanitize(text)
    return f"<user_data>{cleaned}</user_data>"

Q19. RAG retrieval + rerank

def rag_retrieve(question, index, embed_model, rerank_model, k=10, top=3):
    # Vector search
    q_emb = embed_model.encode(question)
    scores, indices = index.search(q_emb.reshape(1, -1), k)
    candidates = [(docs[i], scores[0][j]) for j, i in enumerate(indices[0])]

    # Rerank with cross-encoder
    pairs = [(question, doc) for doc, _ in candidates]
    rerank_scores = rerank_model.predict(pairs)
    reranked = sorted(zip(candidates, rerank_scores), key=lambda x: -x[1])
    return [doc for (doc, _), _ in reranked[:top]]

Coding interview anti-patterns

Đừng làm khi coding phỏng vấn
  1. Code ngay không hỏi: giả định input format, edge case. Rule: 2 phút clarify → confirm ví dụ → mới code.
  2. Im lặng: interviewer không biết bạn nghĩ gì → assume stuck. Rule: "narrate" — nói to suy nghĩ, kể cả khi sai.
  3. Optimize sớm: viết brute force xong rồi improve. Đừng cố ra O(n) ngay, sẽ bug + hết giờ.
  4. Không test edge: empty, single element, duplicates, negative, overflow. Rule: 3 test cases minimum trước "done".
  5. Từ chối hint: interviewer gợi ý = cứu bạn khỏi wasted time. Nhận + acknowledge + iterate.

System design template — 5 phần cho 45 phút

1. Requirements (5 min)
   - Functional: what does system do? 3-5 use cases
   - Non-functional: QPS, latency SLA, availability, consistency
   - Constraints: budget, existing infra, team size

2. API + data model (5 min)
   - REST endpoints or gRPC methods
   - Schema (SQL tables or protobuf)
   - Sample request/response

3. High-level architecture (10 min)
   - Boxes: client, API gateway, service, DB, cache, queue
   - Arrows: data flow
   - Identify 1-2 components để deep-dive

4. Deep-dive (15 min)
   - Pick riskiest / most interesting component
   - Algorithm, data structure, sharding strategy

5. Scale + failure (10 min)
   - 10× traffic — bottleneck ở đâu, mitigate?
   - 5 failure modes + mitigations
   - Monitoring + alerting basics
